    Interface BackendOperations

    Maps each backend operation to its request and response payload types. These mirror the BackendAdapter contract exactly.

    interface BackendOperations {
        chatWithDocument: {
            request: {
                chatHistory?: { role: string; text: string }[];
                guid: string;
                question: string;
            };
            response: ChatResponseResult;
        };
        deleteDocument: {
            request: { guid: string };
            response: { message: string; success: boolean };
        };
        extractMetadata: {
            request: {
                fileData?: string;
                fileName: string;
                mimeType?: string;
                sampleContent?: string;
            };
            response: BibTeXMetadata & { extractedText?: string };
        };
        getDocument: { request: { guid: string }; response: DocumentRecord };
        getDocuments: {
            request: {
                filters: FilterState;
                page: number;
                pageSize: number;
                sortBy: string;
                sortOrder: "desc" | "asc";
            };
            response: DocumentListResult;
        };
        getOpenApiSpec: { request: Record<string, never>; response: string };
        getVersionHistory: {
            request: { guid: string };
            response: DocumentVersionSnapshot[];
        };
        login: {
            request: { password?: string; realm?: string; username: string };
            response: { accessToken: string; user: UserProfile };
        };
        logout: { request: Record<string, never>; response: void };
        overwriteVersion: {
            request: {
                bibtex?: BibTeXMetadata;
                file?: File | null;
                fileContent?: string;
                fileData?: string;
                fileFormat?: string;
                fileName?: string;
                fileSize?: number;
                guid: string;
                mimeType?: string;
            };
            response: DocumentRecord;
        };
        regenerateSummary: {
            request: { guid: string; modelKey: "llama"
            | "mistral" };
            response: SummaryRecord;
        };
        rollbackVersion: {
            request: { guid: string; targetVersion: number };
            response: DocumentRecord;
        };
        testHealth: {
            request: Record<string, never>;
            response: BackendHealthResult;
        };
        uploadDocument: {
            request: {
                bibtex: BibTeXMetadata;
                file: File;
                fileContent?: string;
                fileData?: string;
                fileFormat: string;
                fileName: string;
                fileSize: number;
                mimeType?: string;
            };
            response: DocumentRecord;
        };
    }
